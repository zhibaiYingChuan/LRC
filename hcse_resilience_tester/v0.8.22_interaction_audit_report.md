  const MEMORY_WARN_THRESHOLD = 0.8;
  const MEMORY_CHECK_INTERVAL = 30000;

  function init() {
    // 1. 全局错误处理（v0.8.22 IA-02 已修复，此处作为兜底）
    if (!window._lrcGlobalErrorRegistered) {
      window._lrcGlobalErrorRegistered = true;
      window.addEventListener('error', (event) => {
        console.error('[InteractionGuard] 全局错误:', event.error || event.message);
        if (typeof showToast === 'function') {
          showToast('发生未知错误，请刷新页面', 'error', 5000);
        }
        // 上报到 sidecar
        try {
          fetch('/v1/log', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              level: 'error',
              source: 'InteractionGuard.onerror',
              message: event.message,
              filename: event.filename,
              lineno: event.lineno,
              colno: event.colno,
              stack: event.error && event.error.stack,
            }),
          }).catch(() => {});
        } catch (e) {}
      });
      window.addEventListener('unhandledrejection', (event) => {
        console.error('[InteractionGuard] 未捕获 Promise:', event.reason);
        if (typeof showToast === 'function') {
          const msg = (event.reason && event.reason.message) || '操作失败，请重试';
          showToast(msg, 'error', 3000);
        }
      });
      console.log('[InteractionGuard] 全局错误处理已注册');
    }

    // 2. 快速点击防护（修复 IA-22-05/IA-10）
    document.addEventListener(
      'click',
      (event) => {
        const target = event.target.closest('button, [role="button"], .btn');
        if (!target) return;
        const key = target.dataset.action || target.id || target.textContent.trim().substring(0, 30);
        const now = Date.now();
        const timestamps = _clickCounters.get(key) || [];
        const recent = timestamps.filter((t) => now - t < 1000);
        if (recent.length >= MAX_CLICKS_PER_SECOND) {
          console.warn('[InteractionGuard] 检测到快速点击: ' + key + ' (' + recent.length + '/s)');
          event.preventDefault();
          event.stopPropagation();
          if (typeof showToast === 'function') {
            showToast('操作过于频繁，请稍后再试', 'warning', 2000);
          }
          return;
        }
        recent.push(now);
        _clickCounters.set(key, recent);
      },
      true
    );

    // 3. Modal 栈管理（修复 IA-22-10/IA-15 Z-index 错乱）
    const observer = new MutationObserver(() => {
      const modals = Array.from(document.querySelectorAll('.modal')).filter(
        (m) => m.style.display !== 'none' && getComputedStyle(m).display !== 'none'
      );
      if (modals.length > 1) {
        console.warn('[InteractionGuard] 检测到 ' + modals.length + ' 个 modal 同时打开');
        modals.slice(0, -1).forEach((m) => {
          m.style.display = 'none';
        });
      }
    });
    observer.observe(document.body, { attributes: true, subtree: true, attributeFilter: ['style'] });

    // 4. 内存监控（修复 IA-22-08/IA-13）
    if (performance.memory) {
      setInterval(() => {
        const used = performance.memory.usedJSHeapSize;
        const limit = performance.memory.jsHeapSizeLimit;
        const ratio = used / limit;
        if (ratio > MEMORY_WARN_THRESHOLD) {
          console.warn('[InteractionGuard] 内存使用率 ' + Math.round(ratio * 100) + '% (' + Math.round(used / 1024 / 1024) + 'MB/' + Math.round(limit / 1024 / 1024) + 'MB)');
          if (typeof showToast === 'function' && !window._memWarnShown) {
            window._memWarnShown = true;
            showToast('内存使用过高，建议刷新页面', 'warning', 5000);
            setTimeout(() => (window._memWarnShown = false), 60000);
          }
        }
      }, MEMORY_CHECK_INTERVAL);
    }

    // 5. 暴露闭包内变量到 window 便于 CDP 测试（修复 IA-22-02/IA-22-15）
    // 注意：必须在 app.js IIFE 内部执行
    try {
      if (typeof daoAbortController !== 'undefined') {
        Object.defineProperty(window, '__daoAbortController', {
          get: () => daoAbortController,
          configurable: true,
        });
      }
      if (typeof dashboardAbortController !== 'undefined') {
        Object.defineProperty(window, '__dashboardAbortController', {
          get: () => dashboardAbortController,
          configurable: true,
        });
      }
      if (typeof _tabAbortControllers !== 'undefined') {
        Object.defineProperty(window, '__tabAbortControllers', {
          get: () => _tabAbortControllers,
          configurable: true,
        });
      }
    } catch (e) {
      console.warn('[InteractionGuard] 暴露闭包变量失败:', e);
    }

    console.log('[InteractionGuard] 已初始化（全局错误+快速点击+modal栈+内存监控+CDP测试支持）');
  }

  return { init, _modalStack };
})();

// 自动初始化（DOM 加载后，必须在 app.js IIFE 内部调用）
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', InteractionGuard.init);
} else {
  InteractionGuard.init();
